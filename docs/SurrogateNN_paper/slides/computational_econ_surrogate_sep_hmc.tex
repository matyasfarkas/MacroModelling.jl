\begin{frame}{Main Takeaway First}
\claim{Do not put the global nonlinear solver inside the posterior sampler.}

\vspace{1.1em}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.48\textwidth}
\textbf{What I do}

\softline
I solve the nonlinear model offline with Stochastic Extended Path, train a
differentiable residual surrogate, and use that surrogate inside HMC.
\end{column}
\begin{column}{0.48\textwidth}
\textbf{Why it matters}

\softline
This turns nonlinear solution cost from a per-likelihood cost into an
offline training and validation cost.
\end{column}
\end{columns}

\smallnote{The contribution is computational: make Bayesian nonlinear DSGE estimation operational without pretending the nonlinear solver is cheap.}
\end{frame}

% -----------------------------------------------------------------------------
\begin{frame}{The Hook}
\claim{Nonlinear DSGE models are economically natural and computationally awkward.}

\vspace{0.5em}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.50\textwidth}
\textbf{Economists care about nonlinear states.}
\begin{itemize}
  \item Effective lower bounds.
  \item Occasionally binding constraints.
  \item Large shocks and crisis periods.
  \item Curvature in pricing, wages, investment, and utilization.
\end{itemize}
\end{column}
\begin{column}{0.46\textwidth}
\textbf{But estimation usually retreats to linear tools.}
\begin{itemize}
  \item Perturbation is fast.
  \item Kalman filtering is stable.
  \item The full nonlinear likelihood is too expensive.
  \item Particle filters can be fragile in high dimension.
\end{itemize}
\end{column}
\end{columns}

\vspace{0.55em}
\begin{center}
\normalsize The question is not whether nonlinearities matter. The question is whether we can estimate them at scale.
\end{center}
\end{frame}

% -----------------------------------------------------------------------------
\begin{frame}{Question}
\claim{How can we estimate nonlinear DSGE models without solving the model at every posterior draw?}

\vspace{0.8em}
\textbf{The empirical problem}

\softline
Medium-scale DSGE models are useful because they connect mechanisms, shocks,
and observables. The same structure makes nonlinear estimation expensive.

\vspace{0.7em}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.48\textwidth}
\textbf{What we want}
\begin{itemize}
  \item Structural parameters with posterior uncertainty.
  \item Likelihood comparisons across nonlinear mechanisms.
  \item Diagnostics that say when linearization is misleading.
\end{itemize}
\end{column}
\begin{column}{0.48\textwidth}
\textbf{What blocks us}
\begin{itemize}
  \item Repeated nonlinear solves.
  \item Numerical failures away from the mode.
  \item Slow or noisy likelihood evaluations.
\end{itemize}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Step 0: Choose the Estimation Target}
\claim{Be explicit about the object before approximating it.}

\vspace{0.5em}
Start with a nonlinear model, a parameter vector \(\theta\), observed data
\(\obs_{1:T}\), and a likelihood or profiled likelihood criterion.

\vspace{0.4em}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.48\textwidth}
\textbf{Inputs}
\begin{itemize}
  \item Model equations.
  \item Parameter support.
  \item Shock process and observables.
  \item Measurement-error or inversion design.
\end{itemize}
\end{column}
\begin{column}{0.48\textwidth}
\textbf{Audit trail}
\begin{itemize}
  \item Which parameters vary?
  \item What region is trained?
  \item What likelihood is approximated?
  \item What is held fixed?
\end{itemize}
\end{column}
\end{columns}
\end{frame}

% -----------------------------------------------------------------------------
\begin{frame}{Step 1: Solve the Nonlinear Model Offline}
\claim{Use SEP as an offline teacher.}

\vspace{0.4em}
For sampled triples \((\state,\eps,\theta)\), compute the nonlinear transition
\[
  \state^+_{\sep} = g^{\sep}_{\theta}(\state,\eps).
\]

\vspace{0.4em}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.48\textwidth}
\textbf{Why SEP?}
\begin{itemize}
  \item Keeps the nonlinear equations.
  \item Handles occasionally binding constraints.
  \item Gives direct objects for validation.
\end{itemize}
\end{column}
\begin{column}{0.48\textwidth}
\textbf{What must be reported}
\begin{itemize}
  \item Solver tolerances.
  \item Failed and successful points.
  \item Shock and parameter support.
  \item Initialization strategy.
\end{itemize}
\end{column}
\end{columns}
\end{frame}

% -----------------------------------------------------------------------------
\begin{frame}{Step 2: Compute the Linear Baseline at the Same Points}
\claim{Use the linear solution as the baseline, not as the answer.}

\vspace{0.4em}
At the same inputs, compute
\[
  \state^+_{\rom} = g^{\rom}_{\theta}(\state,\eps).
\]

\vspace{0.4em}
\textbf{Reason}

\softline
The linear solution is cheap and often accurate near the steady state. The NN
should not waste capacity relearning what ROM1 already does well.

\vspace{0.4em}
\begin{center}
\large
\[
  \text{Nonlinear object}
  =
  \text{linear baseline}
  +
  \text{nonlinear correction}.
\]
\end{center}
\end{frame}

% -----------------------------------------------------------------------------
\begin{frame}{Step 3: Train the Residual Surrogate}
\claim{Train the nonlinear correction.}

\vspace{0.4em}
Train \(r_{\phi}\) on the residual
\[
  r_{\phi}(\state,\eps,\theta)
  \approx
  g^{\sep}_{\theta}(\state,\eps)
  -
  g^{\rom}_{\theta}(\state,\eps).
\]

\vspace{0.5em}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.48\textwidth}
\textbf{Why residual learning}
\begin{itemize}
  \item Smaller target.
  \item Easier extrapolation around the linear region.
  \item Directly measures what linearization misses.
\end{itemize}
\end{column}
\begin{column}{0.48\textwidth}
\textbf{Resulting transition}
\[
  g_{\phi}(\state,\eps,\theta)
  =
  g^{\rom}_{\theta}(\state,\eps)
  +
  r_{\phi}(\state,\eps,\theta).
\]
\end{column}
\end{columns}
\end{frame}

% -----------------------------------------------------------------------------
\begin{frame}{Step 4: Validate the Surrogate Before Estimation}
\claim{Do not estimate before the surrogate passes basic audits.}

\vspace{0.25em}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.48\textwidth}
\includegraphics[width=\textwidth,height=0.56\textheight,keepaspectratio]{zlb_rrmse_comparison.png}
\end{column}
\begin{column}{0.48\textwidth}
\textbf{Checks}
\begin{itemize}
  \item Held-out direct SEP points.
  \item RMSE and relative RMSE by observable.
  \item Regime-specific accuracy: binding, non-binding, near-boundary.
  \item Finite likelihood and finite-gradient checks.
\end{itemize}
\vspace{0.25em}
\textbf{Rule}

\softline
The surrogate is only credible on the support where it has been audited.
\end{column}
\end{columns}
\end{frame}

% -----------------------------------------------------------------------------
\begin{frame}{Step 5: Build the Likelihood}
\claim{The likelihood determines what the validation has to prove.}

\vspace{0.4em}
\textbf{Small validation model}

\softline
Use a direct measurement-error likelihood. Compare direct SEP-HMC with
surrogate-HMC on the same synthetic data.

\vspace{0.4em}
\textbf{Medium-scale empirical model}

\softline
Use an inversion step to recover shocks period by period, then propagate with
\[
  \widehat{\state}_{t+1}
  =
  g^{\rom}_{\theta}(\state_t,\widehat{\eps}_{t+1})
  +
  r_{\phi}(\state_t,\widehat{\eps}_{t+1},\theta).
\]

\smallnote{The medium-scale target is a profiled hybrid criterion. That caveat should be explicit.}
\end{frame}

% -----------------------------------------------------------------------------
\begin{frame}{Step 6: Sample Parameters with HMC}
\claim{The differentiable surrogate is what makes HMC practical.}

\vspace{0.4em}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.50\textwidth}
\textbf{Why HMC}
\begin{itemize}
  \item Uses gradients to move through correlated DSGE posteriors.
  \item Needs many fast target evaluations.
  \item Works naturally with differentiable surrogates.
\end{itemize}
\end{column}
\begin{column}{0.46\textwidth}
\textbf{What is matched in validation}
\begin{itemize}
  \item Same data.
  \item Same priors.
  \item Same transforms.
  \item Same seeds and HMC settings.
\end{itemize}
\end{column}
\end{columns}

\vspace{0.55em}
\textbf{Comparison}

\softline
The surrogate posterior should match the direct SEP posterior within Monte
Carlo error, with no systematic directional distortion.
\end{frame}

% -----------------------------------------------------------------------------
\begin{frame}{Step 7: Audit the Whole Pipeline}
\claim{A surrogate paper lives or dies by its audit trail.}

\vspace{0.35em}
\begin{table}
\centering
\small
\begin{tabular}{lll}
\toprule
Object & Question & Failure mode \\
\midrule
SEP support & Did the solver cover the region? & Hidden selection \\
Surrogate fit & Is the map accurate where used? & Biased likelihood \\
Gradients & Are target derivatives finite? & Bad HMC geometry \\
HMC chains & Do chains mix and agree? & Unreliable posterior \\
Direct audits & Does SEP confirm key cells? & Unsupported extrapolation \\
\bottomrule
\end{tabular}
\end{table}

\vspace{0.4em}
\begin{center}
\small The replication package should make these checks mechanical, not narrative.
\end{center}
\end{frame}

% -----------------------------------------------------------------------------
\begin{frame}{The Complete Pipeline}
\claim{The method is a sequence of auditable approximations.}

\vspace{0.25em}
\begin{enumerate}
  \item Define model, observables, parameter support, and likelihood target.
  \item Draw training states, shocks, and parameters.
  \item Solve direct nonlinear transitions with SEP.
  \item Compute ROM1 transitions at identical inputs.
  \item Train the NN on the SEP--ROM1 residual.
  \item Validate the map on held-out direct SEP points.
  \item Insert \(g^{\rom}+r_{\phi}\) into the likelihood.
  \item Run HMC and audit the posterior against direct SEP where feasible.
\end{enumerate}

\vspace{0.25em}
\textbf{The discipline}

\softline
Every empirical number should trace back to raw artifacts: support, errors,
diagnostics, and sampler output.
\end{frame}

\begin{frame}{Practical Lessons}
\claim{The practical lesson is discipline, not automation.}

\vspace{0.35em}
\begin{enumerate}
  \item \textbf{Train the correction, not the whole model.}
  ROM1-residual learning is easier to audit than a direct map.
  \item \textbf{Report solver coverage.}
  Failed SEP points are information, not bookkeeping noise.
  \item \textbf{Validate where the posterior lives.}
  Average approximation accuracy is not enough.
  \item \textbf{Keep the empirical claim conditional.}
  Medium-scale results are only as strong as the validated support.
\end{enumerate}
\end{frame}

\begin{frame}{Closing Takeaways}
\claim{The method is promising because it is auditable step by step.}

\vspace{0.35em}
\begin{enumerate}
  \item The computational obstacle is clear: global nonlinear solves are too
        expensive for the inner likelihood loop.
  \item The proposed solution is simple: use SEP offline, learn the ROM1 residual,
        and put a differentiable surrogate inside HMC.
  \item The small nonlinear validation supports the pipeline.
  \item The Smets--Wouters extension shows where the hard work remains:
        solver stability, surrogate error, and posterior-region coverage.
\end{enumerate}

\smallnote{The remaining work is clear: more robust SEP initialization, tighter posterior-region audits, and broader direct benchmarks.}
\end{frame}

\begin{frame}{Backup: Target and Surrogate Error}
\[
  e_{\phi}(\state,\eps,\theta)
  =
  g^{\sep}_{\theta}(\state,\eps)
  -
  \left[
    g^{\rom}_{\theta}(\state,\eps)
    +
    r_{\phi}(\state,\eps,\theta)
  \right].
\]

\vspace{0.5em}
\begin{block}{Validation quantity}
\[
  \mathrm{RMSE}_{\mathrm{val}}
  =
  \left(
  \frac{1}{N}\sum_{i=1}^{N}
  \| e_{\phi}(\state_i,\eps_i,\theta_i) \|^2
  \right)^{1/2}.
\]
\end{block}

\begin{center}
\small The same held-out inputs are evaluated under direct SEP, ROM1, and the surrogate.
\end{center}
\end{frame}

\begin{frame}{Backup: Matched HMC Validation Design}
\begin{itemize}
  \item Same synthetic data.
  \item Same prior and parameter transforms.
  \item Same initial values and random seeds.
  \item Same NUTS settings: warmup, draws, target acceptance, tree depth.
  \item Compare posterior means, MCSEs, and 90 percent intervals.
\end{itemize}

\vspace{0.5em}
\begin{block}{Pass criterion}
The surrogate posterior should overlap the direct SEP posterior within Monte
Carlo uncertainty, with no systematic directional distortion.
\end{block}
\end{frame}

\begin{frame}{Backup: Why SEP?}
\begin{itemize}
  \item Keeps the original nonlinear equations.
  \item Handles occasionally binding constraints directly.
  \item Produces a simulation object that can be compared with perturbation.
  \item Expensive, but parallel over training points.
\end{itemize}

\vspace{0.5em}
\begin{block}{Tradeoff}
SEP is a good offline teacher, but a poor online likelihood engine.
\end{block}
\end{frame}

% -----------------------------------------------------------------------------
\begin{frame}{Backup: Posterior Interpretation}
\begin{itemize}
  \item The small-model validation targets direct likelihood comparisons.
  \item The medium-scale empirical exercise uses an inversion/profiled criterion.
  \item Surrogate HMC results should be read as criterion-based posterior
        diagnostics until a full direct-SEP HMC benchmark is feasible.
\end{itemize}

\vspace{0.5em}
\begin{center}
\small The method is promising because the validation pieces work, not because every medium-scale empirical caveat is already solved.
\end{center}
\end{frame}
